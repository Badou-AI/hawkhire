# Commands to fix Next.js CSS loading issue

# 1. Update the Nginx configuration for better handling of Next.js static assets
cp /etc/nginx/sites-available/beta.hawkhire.ai /etc/nginx/sites-available/beta.hawkhire.ai.bak2

# 2. Create the updated configuration with proper handling of /_next/static
cat > /etc/nginx/sites-available/beta.hawkhire.ai << 'EOF'
server {
    server_name beta.hawkhire.ai;
    
    # Include security headers
    include /etc/nginx/snippets/security-headers.conf;
   
    # Important: Improved handling of Next.js static assets
    location /_next/static/ {
        root /var/www/beta.hawkhire.ai;
        try_files $uri $uri/ =404;
        add_header Cache-Control "public, max-age=31536000, immutable";
    }
    
    # Handle static files in public directory directly
    location ~ \.(html|css|js|svg|ico|jpg|jpeg|png|gif)$ {
        root /var/www/beta.hawkhire.ai/public;
        add_header Cache-Control "public, max-age=31536000, immutable";
        try_files $uri =404;
    }
    
    # Handle other static files
    location /static/ {
        root /var/www/beta.hawkhire.ai/public;
        add_header Cache-Control "public, max-age=31536000, immutable";
    }
    
    # Handle public files
    location /public/ {
        root /var/www/beta.hawkhire.ai;
        add_header Cache-Control "public, max-age=31536000, immutable";
    }
    
    # Handle favicon and other root files
    location ~ ^/(favicon\.ico|robots\.txt|sitemap\.xml)$ {
        root /var/www/beta.hawkhire.ai/public;
    }
    
    # Proxy all other requests to the Next.js server
    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;      
        proxy_set_header Connection 'upgrade';       
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;     
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;  
        proxy_cache_bypass $http_upgrade;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/www.hawkhire.ai/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/www.hawkhire.ai/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}

server {
    if ($host = beta.hawkhire.ai) {
        return 301 https://$host$request_uri;        
    } # managed by Certbot

    listen 80;
    server_name beta.hawkhire.ai;
    return 404; # managed by Certbot
}
EOF

# 3. Test the configuration
nginx -t

# 4. If the test is successful, restart Nginx
systemctl restart nginx

# 5. Create symbolic links to ensure CSS files are accessible
mkdir -p /var/www/beta.hawkhire.ai/public/_next/static/css/
ln -sf /var/www/beta.hawkhire.ai/.next/static/css/* /var/www/beta.hawkhire.ai/public/_next/static/css/
